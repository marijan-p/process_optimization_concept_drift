\renewcommand*{\acsfont}[1]{\texttt{#1}}

\makeatletter
\renewcommand*{\@acf}[2][\AC@linebreakpenalty]{%
   \ifAC@footnote
      \acsfont{\AC@acs{#2}}%
      \footnote{\AC@placelabel{#2}\AC@acl{#2}{}}%
   \else
      \acffont{%
         \AC@placelabel{#2}\AC@acl{#2}%
         \nolinebreak[#1] %
         \acfsfont{, \acsfont{\AC@acs{#2}}}%
      }%
   \fi
   \ifAC@starred\else\AC@logged{#2}\fi}
